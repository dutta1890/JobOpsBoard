<script>
function startAgent() {
    fetch('/agent/start', {method: 'POST'}).then(updateAgentUI);
}

function stopAgent() {
    fetch('/agent/stop', {method: 'POST'}).then(updateAgentUI);
}

function queueGradeA() {
    fetch('/agent/queue-grade-a', {method: 'POST'})
        .then(r => r.json())
        .then(d => {
            alert('Queued ' + d.queued + ' Grade A jobs');
            updateAgentUI();
        });
}

function queueGradeB() {
    fetch('/agent/queue-grade-b', {method: 'POST'})
        .then(r => r.json())
        .then(d => {
            alert('Queued ' + d.queued + ' Grade B jobs');
            updateAgentUI();
        });
}

function resumeTask(taskId) {
    fetch('/agent/resume/' + taskId, {method: 'POST'}).then(updateAgentUI);
}

function updateAgentUI() {
    fetch('/agent/status')
        .then(r => r.json())
        .then(data => {
            // Badge
            const badge = document.getElementById('agent-badge');
            if (data.is_running) {
                badge.textContent = 'Running';
                badge.className = 'px-2 py-1 text-xs rounded bg-green-900 text-green-300';
                document.getElementById('start-btn').classList.add('hidden');
                document.getElementById('stop-btn').classList.remove('hidden');
            } else {
                badge.textContent = 'Stopped';
                badge.className = 'px-2 py-1 text-xs rounded bg-gray-700 text-gray-400';
                document.getElementById('start-btn').classList.remove('hidden');
                document.getElementById('stop-btn').classList.add('hidden');
            }
            
            // Stats
            document.getElementById('q-queued').textContent = data.queue.queued;
            document.getElementById('q-processing').textContent = data.queue.processing;
            document.getElementById('q-waiting').textContent = data.queue.waiting_auth;
            document.getElementById('q-done').textContent = data.stats.successful;
            document.getElementById('q-failed').textContent = data.stats.failed;
            
            // Current task
            const ct = document.getElementById('current-task');
            if (data.current) {
                ct.classList.remove('hidden');
                document.querySelector('#current-text span:last-child').textContent = 
                    data.current.company + ' - ' + data.current.role;
            } else {
                ct.classList.add('hidden');
            }
            
            // Auth waiting
            const authSection = document.getElementById('auth-section');
            const authList = document.getElementById('auth-list');
            if (data.waiting_auth && data.waiting_auth.length > 0) {
                authSection.classList.remove('hidden');
                authList.innerHTML = data.waiting_auth.map(t => `
                    <div class="bg-yellow-900/30 border border-yellow-700 rounded p-3 flex justify-between items-center">
                        <div>
                            <div class="text-white">${t.company} - ${t.role}</div>
                            <div class="text-yellow-400 text-xs">Login to ${t.portal}</div>
                        </div>
                        <button onclick="resumeTask(${t.id})"
                            class="bg-green-600 px-3 py-1 rounded text-xs">
                            Done ✓
                        </button>
                    </div>
                `).join('');
            } else {
                authSection.classList.add('hidden');
            }
        });
}

// Auto-refresh every 3 seconds
setInterval(updateAgentUI, 3000);
updateAgentUI();
</script>


## TASK 2: Add queue-grade-b route

Add to main.py:

@app.route('/agent/queue-grade-b', methods=['POST'])
def queue_all_grade_b():
    from services.agent import agent
    from models.database import Application
    
    apps = Application.query.filter_by(grade='B', status='Evaluated').all()
    
    count = 0
    for app in apps:
        agent.add_task(app.id, app.company, app.role, app.url, app.jd_text)
        count += 1
    
    return jsonify({'queued': count})
