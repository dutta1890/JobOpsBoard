"""
Email Applicator - 100% Automated
Sends job application emails without human intervention
"""

import smtplib
import os
import re
from email.mime.multipart import MIMEMultipart
from email.mime.text import MIMEText
from email.mime.base import MIMEBase
from email import encoders
from datetime import datetime
from pathlib import Path

class EmailApplicator:
    """Sends application emails automatically."""
    
    def __init__(self):
        self.smtp_server = os.environ.get('SMTP_SERVER', 'smtp.gmail.com')
        self.smtp_port = int(os.environ.get('SMTP_PORT', 587))
        self.email_address = os.environ.get('EMAIL_ADDRESS', '')
        self.email_password = os.environ.get('EMAIL_APP_PASSWORD', '')
        
        # Your details
        self.sender_name = "Aishik Dutta"
        self.phone = "+91-8741-03-4164"
        self.linkedin = "linkedin.com/in/aishikduttaoct18"
        
        # CV path
        self.cv_path = Path('data/cv.pdf')
    
    def apply(self, to_email, company, role, cover_letter_text):
        """
        Send application email.
        Returns: {success: bool, error: str, timestamp: str}
        """
        if not to_email:
            return {'success': False, 'error': 'No email address'}
        
        if not self.email_address or not self.email_password:
            return {'success': False, 'error': 'Email credentials not configured'}
        
        try:
            # Build email
            msg = MIMEMultipart()
            msg['From'] = f"{self.sender_name} <{self.email_address}>"
            msg['To'] = to_email
            msg['Subject'] = f"Application for {role} - {self.sender_name}"
            
            # Body
            body = f"""Dear Hiring Manager,

{cover_letter_text}

Best regards,
{self.sender_name}
{self.phone}
{self.linkedin}

---
Resume attached.
"""
            msg.attach(MIMEText(body, 'plain'))
            
            # Attach CV
            if self.cv_path.exists():
                with open(self.cv_path, 'rb') as f:
                    attachment = MIMEBase('application', 'pdf')
                    attachment.set_payload(f.read())
                    encoders.encode_base64(attachment)
                    attachment.add_header(
                        'Content-Disposition',
                        f'attachment; filename="{self.sender_name.replace(" ", "_")}_CV.pdf"'
                    )
                    msg.attach(attachment)
            
            # Send
            with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
                server.starttls()
                server.login(self.email_address, self.email_password)
                server.send_message(msg)
            
            return {
                'success': True,
                'method': 'email',
                'to': to_email,
                'timestamp': datetime.now().isoformat()
            }
            
        except Exception as e:
            return {'success': False, 'error': str(e)}
    
    def extract_email_from_jd(self, jd_text):
        """Extract email from JD text."""
        pattern = r'[\w\.-]+@[\w\.-]+\.\w+'
        matches = re.findall(pattern, jd_text)
        
        # Filter false positives
        valid = [e for e in matches if not any(x in e.lower() for x in [
            'example.com', 'email.com', 'domain.com', 'company.com'
        ])]
        
        return valid[0] if valid else None
    
    def test_connection(self):
        """Test SMTP connection."""
        try:
            with smtplib.SMTP(self.smtp_server, self.smtp_port) as server:
                server.starttls()
                server.login(self.email_address, self.email_password)
            return True
        except Exception as e:
            print(f"Connection failed: {e}")
            return False
