# Rate limiting (Gemini free tier: 60/min)
LAST_REQUEST = 0
MIN_INTERVAL = 1.0

def rate_limited(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        global LAST_REQUEST
        elapsed = time.time() - LAST_REQUEST
        if elapsed < MIN_INTERVAL:
            time.sleep(MIN_INTERVAL - elapsed)
        result = func(*args, **kwargs)
        LAST_REQUEST = time.time()
        return result
    return wrapper

# Filter configuration
JOB_AGE_SKIP_DAYS = 7
JOB_AGE_PRIORITY_DAYS = 3

SKIP_TITLES = [
    'intern', 'internship', 'trainee', 'fresher',
    'analyst', 'executive', 'associate', 'junior',
    'entry level', 'entry-level', 'graduate'
]

SKIP_EXPERIENCE = [
    '0-1', '1-2', '2-3', '0-2', '1-3', '0-3',
    'fresher', 'entry level'
]


def check_job_age(posted_date_str):
    """
    Check if job should be skipped based on age.
    Returns: {skip: bool, reason: str, priority: str, days_old: int}
    """
    if not posted_date_str:
        return {'skip': False, 'priority': 'medium', 'days_old': None}
    
    text = posted_date_str.lower()
    days_old = None
    
    # Parse different formats
    if 'just now' in text or 'today' in text:
        days_old = 0
    elif 'yesterday' in text:
        days_old = 1
    elif match := re.search(r'(\d+)\s*hour', text):
        days_old = 0
    elif match := re.search(r'(\d+)\s*day', text):
        days_old = int(match.group(1))
    elif match := re.search(r'(\d+)\s*week', text):
        days_old = int(match.group(1)) * 7
    elif match := re.search(r'(\d+)\s*month', text):
        days_old = int(match.group(1)) * 30
    
    if days_old is None:
        return {'skip': False, 'priority': 'medium', 'days_old': None}
    
    # Apply rules
    if days_old >= JOB_AGE_SKIP_DAYS:
        return {
            'skip': True,
            'reason': f'Job is {days_old} days old (limit: {JOB_AGE_SKIP_DAYS})',
            'priority': 'skip',
            'days_old': days_old
        }
    
    priority = 'high' if days_old <= JOB_AGE_PRIORITY_DAYS else 'medium'
    return {'skip': False, 'priority': priority, 'days_old': days_old}


def check_seniority(title, experience_text='', description=''):
    """
    Check if job is too junior.
    Returns: {skip: bool, reason: str}
    """
    combined = f"{title} {experience_text} {description}".lower()
    
    # Check title
    title_lower = title.lower() if title else ''
    for skip_word in SKIP_TITLES:
        if skip_word in title_lower:
            return {'skip': True, 'reason': f'Title contains "{skip_word}"'}
    
    # Check experience patterns
    for pattern in SKIP_EXPERIENCE:
        if pattern in combined:
            return {'skip': True, 'reason': f'Junior experience: {pattern}'}
    
    # Check numeric patterns
    junior_patterns = [
        r'0\s*[-–]\s*[12]\s*years?',
        r'1\s*[-–]\s*[23]\s*years?',
        r'2\s*[-–]\s*3\s*years?',
    ]
    for pattern in junior_patterns:
        if re.search(pattern, combined):
            return {'skip': True, 'reason': 'Junior experience requirement'}
    
    return {'skip': False}


def pre_filter_job(job_data):
    """
    Run all filters before AI evaluation.
    Returns: {proceed: bool, skip_reason: str, priority: str, days_old: int}
    """
    # Filter 1: Job age
    age_check = check_job_age(job_data.get('posted_date', ''))
    if age_check['skip']:
        return {
            'proceed': False,
            'skip_reason': age_check['reason'],
            'priority': 'skip',
            'filter_type': 'job_age'
        }
    
    # Filter 2: Seniority
    seniority_check = check_seniority(
        job_data.get('title', ''),
        job_data.get('experience_required', ''),
        job_data.get('description', '')
    )
    if seniority_check['skip']:
        return {
            'proceed': False,
            'skip_reason': seniority_check['reason'],
            'priority': 'skip',
            'filter_type': 'seniority'
        }
    
    return {
        'proceed': True,
        'skip_reason': None,
        'priority': age_check['priority'],
        'days_old': age_check.get('days_old')
    }

@rate_limited
def evaluate_job(jd_text, job_metadata=None):
    """Evaluate a job using Gemini API."""
    
    # Run pre-filters first
    if job_metadata:
        pre_filter = pre_filter_job(job_metadata)
        if not pre_filter['proceed']:
            return {
                'skip': True,
                'skip_reason': pre_filter['skip_reason'],
                'filter_type': pre_filter['filter_type'],
                'score': 0,
                'grade': 'F',
                'recommendation': 'Skip'
            }
    
    prompt = f"""{EVALUATE_PROMPT}

## JOB DESCRIPTION:
{jd_text}

Output valid JSON only."""
    
    try:
        response = model.generate_content(prompt)
        result = parse_response(response.text)
        
        # Add priority from filters
        if job_metadata:
            pre_filter = pre_filter_job(job_metadata)
            result['priority'] = pre_filter.get('priority', 'medium')
            result['days_old'] = pre_filter.get('days_old')
        
        return result
    except Exception as e:
        return {'error': str(e)}


def parse_response(text):
    """Parse JSON from Gemini response."""
    text = text.strip()
    if text.startswith('```json'):
        text = text[7:]
    if text.startswith('```'):
        text = text[3:]
    if text.endswith('```'):
        text = text[:-3]
    
    try:
        return json.loads(text.strip())
    except:
        return {'error': 'JSON parse failed', 'raw': text[:500]}
